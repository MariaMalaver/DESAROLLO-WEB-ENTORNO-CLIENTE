    Y SI TODAVIA NO LO ESTA SE CREA UN BOTON QUE PERMITE REOLVERLO, AL PULSARLO SE EJECUTA LA FUNCION
    RESOLVER Y SE LE PASA EL IDENTIFICADOR DEL ELEMENTO
*/
html += "<p class='meta'>hace " + this.minutos() + " min</p>"; 

    if (!this.resuelta) 
    {
      html += "<button type='button' onclick='resolver(" + this.id + ")'>Resolver</button>";
    }

/*
    ESTE METODO CALCULA CUANTOS MINUTOS HAN PASADO DESDE QUE SE CREO UN ELEMETO, ENTONCES SE OBTIENE
    LA HORA ACTUAL CON DATE.NOW SE LE RESTA LA FECHA GUARDADA EN THIS.FECHA Y SE DIVIDE EL RESULTADO
    ENTRE 60.000 YA QUE ESA ES LA CANTIDAD DE MILISEGUNDOS QUE TIENE UN MINUTO, POR ULTIMO SE USA EL
    MATH.FLOOR PARA REDONDEAR HACIA ABAJO Y SE OBTIENE SOLO LOS SEGUNDOS COMPLETOS.
*/

minutos() {
    return Math.floor((Date.now() - this.fecha) / 60000); //CAMBIADO DESDE MATH
  }


/*
    SE CREA UNA VARIABLE LLAMADA PENDIENTES Y LA INICIALIZA A CERO, DEPUE SE USA UN BUCLE FOR PARA 
    RECORRER TODAS LAS INCIDENCIAS DEL ARRAY, DENTRO DE ESTE SE COMPRUEBA CON UN IF SI CADA UNA NO 
    ESTA RESUELTA, SI NO LO ESTA SE AUNMENTA EL CONTADOR PENDIENTES EN UNO Y POR ULTIMO SE USA 
    getElementById PARA BUSCAR EL ELEMENTO RESUMEN DE HTML Y SE MUESTRA EL TOTAL DE INCIDENCIAS Y 
    CUANTAS QUEDAN PENDIENTES
*/

let pendientes = 0;
  for (let i = 0; i < incidencias.length; i++) {
    if (!incidencias[i].resuelta) pendientes++;
  }
  document.getElementById("resumen").textContent = "Total: " + incidencias.length + " · Pendientes: " + pendientes;

/*
    ESTA ES PARA RESOLVER UNA INCIDENCIA CUANDO PULSAMOS EL BOTON RESOLVER, DEPUES PRIMERO RECIBE EL
    IDENTIFICADOR DE LA INCIDENCIA QUE QUIERE RESOLVER, LUEGO USA UN BUCLE FOR PARA RECORRER TODAS LAS
    INCIDENCIAS DEL ARRAY , DENTRO DE ETE USA UN IF PARA COMPROBAR SI EL IDENTIFICADOR DE LA INCIDENCIA 
    ACTUAL COINCIDE CON EL QUE A RECIBIDO. Y CUANDO LO ENCUENTRA LA CORRECTA CAMBIA SU PROPIEDAD RESUEKTA 
    A TRUE INDICANDO Q YA ESTA SOLUCIONADA. POR ULTIMO LLAMA A GUARDAR PARA CONSERVAR OS CAMBIOS Y A 
    PINTAR PARA ACTUALIZAR LA INFO QUE APARECE EN LA PAGINA
*/

function resolver(id) 
{
  for (let i = 0; i < incidencias.length; i++) 
  {
    if (incidencias[i].id === id) 
    {
      incidencias[i].resuelta = true;
    }
  }
  guardar();
  pintar();
}
